<compute>
function compute(s) {
  const dk = s.K[0].length;
  const scores = M.matmul(s.Q, M.transpose(s.K));           // q_i · k_j
  const factor = s.scale ? 1 / Math.sqrt(dk) : 1;
  const scaled = M.scale(scores, factor);
  const W = M.softmaxRows(scaled);                          // each row sums to 1
  const out = M.matmul(W, s.V);                             // o_i = sum_j w_ij v_j
  const i = s.focus;
  const terms = W[i].map((w, j) => s.V[j].map(v => w * v));
  return { dk, factor, scores, scaled, W, out, i, terms, rowSums: M.rowSums(W) };
}
