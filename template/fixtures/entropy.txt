<compute>
function compute(s) {
  const p = M.normalize(s.p.slice(0, s.n));
  const contrib = p.map(x => -M.xlog2x(x));          // 0·log 0 = 0
  const H = M.sum(contrib);
  return { p, contrib, H, Hmax: Math.log2(s.n), labels: p.map((_, i) => 'x' + M.subDigits(i + 1)) };
}
</compute>
