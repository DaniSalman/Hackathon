function compute(s) {
  const dk = s.K[0].length;
  const raw = M.matmul(s.Q, M.transpose(s.K));          // 2 × 3 raw scores
  const scaleFactor = s.scale ? 1 / Math.sqrt(dk) : 1;
  const scores = raw.map(row => row.map(v => v * scaleFactor));
  const weights = M.softmaxRows(scores);                 // each row sums to 1
  const out = M.matmul(weights, s.V);                    // 2 × d_v outputs
  const rowSums = M.rowSums(weights);

  // walk index: (i,j) pairs in row-major order
  const pairs = [];
  for (let i = 0; i < s.Q.length; i++) for (let j = 0; j < s.K.length; j++) pairs.push([i, j]);
  const wi = M.clamp(s.walk | 0, 0, pairs.length - 1);
  const [ci, cj] = pairs[wi];
  const q = s.Q[ci], k = s.K[cj];
  const products = q.map((qk, t) => qk * k[t]);
  const dotVal = M.sum(products);

  return { dk, raw, scores, weights, out, rowSums, pairs, ci, cj, q, k, products, dotVal,
           scaleFactor, wi,
           dRow: pairs.length, rowLabels: ['q1', 'q2'], keyLabels: ['k1', 'k2', 'k3'] };
}
