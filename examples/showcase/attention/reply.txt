<compute>
function compute(s) {
  const Q = s.q, K = s.k, V = s.v;
  const dk = Q[0].length;
  // raw scores s_ij = q_i · k_j
  const scores = Q.map(qi => K.map(kj => M.dot(qi, kj)));
  const sdiv = Math.sqrt(dk);
  // scaled scores
  const scaled = scores.map(row => row.map(x => (s.scale ? x / sdiv : x)));
  // per-row softmax
  const alpha = scaled.map(row => M.softmax(row, 1));
  // output = alpha @ V  (rows of alpha times V)
  const O = alpha.map(row => V[0].map((_, c) => M.sum(row.map((a, j) => a * V[j][c]))));
  const rowSums = alpha.map(row => M.sum(row));
  return { Q, K, V, dk, sdiv, scores, scaled, alpha, O, rowSums,
    labels: Q.map((_, i) => 'q' + M.subDigits(i + 1)),
    keyLabels: K.map((_, j) => 'k' + M.subDigits(j + 1)) };
}
