<compute>
function compute(s) {
  // Each Q/K row is stored at width 4; d_k selects how many leading columns are active.
  const dk = Math.min(s.d_k, 4);
  const q = [s.Q[0].slice(0, dk)];
  const k = s.K.map(row => row.slice(0, dk));
  const v = s.V; // 3 value rows, each of length 2

  // Raw scores s_j = q · k_j
  const raw = k.map(row => M.dot(q[0], row));
  const scale = s.scaleOn ? Math.sqrt(dk) : 1;
  const scaled = raw.map(x => x / scale);

  // Softmax with temperature T (T=1 is the paper's formula)
  const T = s.softmaxTemp;
  const alpha = M.softmax(scaled, T);

  // Weighted blend of the values: z = Σ alpha_j v_j
  const z = M.zeros(v[0].length);
  for (let j = 0; j < alpha.length; j++) {
    for (let c = 0; c < z.length; c++) z[c] += alpha[j] * v[j][c];
  }

  const scoreLabels = ['{k_1}', '{k_2}', '{k_3}'];
  const scaledRow = scaled; // already per-key
  return {
    dk, q, k, v, raw, scale, scaled, alpha, z, scoreLabels,
    rawLabels: scoreLabels,
    weightSum: M.sum(alpha),
    outLabels: ['z₁', 'z₂'],
    uniform: [1/3, 1/3, 1/3]
  };
}
