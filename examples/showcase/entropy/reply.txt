<compute>
function compute(s) {
  const n = Math.round(s.n);
  // Keep only the active outcomes and clamp weights to be safe.
  const w = s.w.slice(0, n).map(v => Math.max(0, Number.isFinite(v) ? v : 0));
  const W = M.sum(w);

  // If every weight is zero, fall back to a uniform distribution so nothing is NaN.
  const allZero = W <= 0;
  const p = allZero ? w.map(() => 1 / w.length) : w.map(v => v / W);

  // Guard the zero term explicitly: 0·log2 0 -> 0 by continuity.
  const contrib = p.map(x => x > 0 ? -x * Math.log2(x) : 0);
  const H = M.sum(contrib);

  // n active = positive probability only, so H <= log2(nActive) holds.
  const nActive = p.filter(x => x > 0).length;
  const Hmax = nActive > 0 ? Math.log2(nActive) : 0;

  return {
    n, w, W, p,
    contrib,               // c_i in bits
    H, Hmax, nActive,
    allZero,
    labels: p.map((_, i) => 'x' + M.subDigits(i + 1)),
    ratio: Hmax > 0 ? H / Hmax : 0,
    effective: Math.pow(2, H)   // "as uncertain as this many equally likely outcomes"
  };
}

function render(s, r, kit) {
  // --- panel 1: the raw weights the learner drags ---
  kit.bars('#s1', r.w, {
    bind: 'w', sym: 'w_i', labels: r.labels, axis: 'outcome',
    decimals: 1, min: 0, max: 10, height: 230,
    title: 'Weights {w_i} you type (total Σw = ' + kit.fmt(r.W, 1) + ')',
    ref: { y: r.W / r.n, label: 'average weight' }
  });

  // --- panel 2: the normalized probabilities ---
  kit.bars('#s1', r.p, {
    bind: 'w', sym: 'p_i', labels: r.labels, axis: 'outcome',
    decimals: 3, min: 0, max: 1, height: 230,
    title: 'Normalized probabilities {p_i} = {w_i} / Σw (they always sum to 1)',
    ref: { y: 1 / r.nActive, label: 'uniform 1/n' }
  });
  kit.formula('#s1',
    r.allZero
      ? 'All weights are zero, so the page falls back to uniform: {p_i} = 1 / {n}.'
      : '{p_i} = (' + r.w.map(v => kit.fmt(v, 1)).join(', ') + ') / ' + kit.fmt(r.W, 1) +
        ' = (' + r.p.map(v => kit.fmt(v, 3)).join(', ') + ')');

  // --- scene 2: contributions and the curve ---
  kit.bars('#s2', r.contrib, {
    sym: 'c_i', labels: r.labels, axis: 'outcome', width: 0.5,
    decimals: 3, min: 0, max: 0.6, height: 230,
    title: 'Share of each outcome, {c_i} = −{p_i} log₂ {p_i} (bits)',
    highlight: r.contrib.map((c, i) => c === 0 ? i : -1).filter(i => i >= 0)
  });
  kit.plot('#s2', {
    width: 0.5, height: 230, domain: [0, 1], yRange: [0, 0.6],
    title: 'Every share sits on the curve −p log₂ p',
    series: [{ fn: x => M.xlog2x(x), sym: 'c_i', label: '−p log₂ p', area: true }],
    points: r.p.map((p, i) => ({ x: p, y: r.contrib[i], label: r.labels[i], sym: 'p_i', axis: 'outcome', index: i })),
    vlines: [{ x: 1 / Math.E, label: 'peak at 1/e ≈ 0.37' }],
    xLabel: 'probability p', yLabel: 'bits'
  });
  kit.note('#s2', 'Closed outcomes with {p_i} = 0 contribute exactly {c_i} = 0 bits (the convention 0 · log₂ 0 = 0).');

  // --- scene 3: the total ---
  const gaugeText = r.nActive === 0
    ? 'no active outcomes'
    : 'H = ' + kit.fmt(r.H, 3) + ' bits of at most ' + kit.fmt(r.Hmax, 3) + ' bits (log₂ ' + r.nActive + ')';
  kit.readout('#s3', {
    width: 0.5, label: 'entropy {H}', value: r.H, unit: 'bits', sym: 'H',
    decimals: 3, min: 0, max: r.Hmax > 0 ? r.Hmax : 1,
    compare: s.logScale ? { value: r.Hmax, label: 'max = log₂ {n} = ' + kit.fmt(r.Hmax, 3) } : undefined,
    note: r.H < 1e-12
      ? 'No uncertainty: one outcome has all the weight.'
      : 'As uncertain as ' + kit.fmt(r.effective, 2) + ' equally likely outcomes.'
  });
  kit.steps('#s3', [
    { label: 'probabilities {p_i}', value: r.p, sym: 'p_i' },
    { label: 'shares {c_i} (bits)', value: r.contrib, sym: 'c_i', op: '−p log₂ p' },

}
