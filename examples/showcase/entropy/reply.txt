<compute>
function compute(s) {
  const n = Math.max(2, Math.round(s.n));
  let w = (s.w || []).slice(0, n).map(x => Math.max(0, (typeof x === 'number' && isFinite(x)) ? x : 0));
  while (w.length < n) w.push(0);
  let S = M.sum(w);
  const allZero = !(S > 0);
  if (allZero) { w = w.map(() => 1); S = M.sum(w); }
  const p = w.map(x => x / S);
  const contrib = p.map(x => -M.xlog2x(x));          // 0·log 0 = 0
  const H = M.sum(contrib);
  const Hmax = Math.log2(n);
  const labels = p.map((_, i) => 'x' + M.subDigits(i + 1));
  const peakAt = 1 / Math.E;
  const peak = -M.xlog2x(peakAt);                    // ≈ 0.5307 bits
  return { n, w, S, p, contrib, H, Hmax, labels, allZero, peak, peakAt };
}
