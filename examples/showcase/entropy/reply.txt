function compute(s) {
  const n = M.round(s.n);
  let w = s.w.slice(0, n).map(x => Math.max(0, x));
  while (w.length < n) w.push(0);
  if (s.uniform) w = w.map(() => 1);

  const total = M.sum(w);
  // All-zero weights: fall back to uniform so we never divide by zero or return NaN.
  const degenerate = total <= 0;
  const p = degenerate ? w.map(() => 1 / n) : M.normalize(w);

  const contrib = p.map(x => -M.xlog2x(x));   // 0·log 0 = 0, and p=1 gives 0
  const H = M.sum(contrib);
  const Hmax = Math.log2(n);
  const pMax = p.length ? M.max(p) : 0;

  return {
    n,
    w,
    total,
    degenerate,
    p,
    contrib,
    H,
    Hmax,
    uniformGap: Hmax - H,
    labels: p.map((_, i) => 'x' + M.subDigits(i + 1)),
    // number of equally likely outcomes H is worth, 2^H
    effective: Math.pow(2, H)
  };
}
